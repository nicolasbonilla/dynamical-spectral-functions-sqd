% =====================================================================================
%  main_jcp.tex  --  the main text and its references as their own PDF, for The Journal of
%  Chemical Physics, which takes the supplementary material as a separate file (sm_jcp.tex).
%
%  This file carries no text of the paper.  It runs main.tex ITSELF -- the same class,
%  the same preamble, the same \input files in the same order -- and ends the document at
%  the point where main.tex opens the Supplemental Material (\input{sm_begin.tex}), that
%  is, right after \input{bibliography.tex}.  So it cannot drift from main.tex, and the
%  guardians, which read every paper/*.tex, find no second copy of any sentence here.
%
%  What it adds to main.tex, and nothing else:
%    * xr-hyper, so that a reference to an SM item prints as in main.pdf ("Sec. S3",
%      "Fig. S2", "Eq. (S12)"), read from sm_jcp.aux.  [nocite]: the labels of the SM
%      are imported, its citations are not (each file numbers its own reference list).
%      xr-hyper has to be loaded before hyperref, hence before the class here.
%    * REVTeX's own bookkeeping labels (FirstPage, LastPage, LastBibItem) are defined by
%      both files; the imported copies are dropped, so no label is multiply defined.
%    * At the stop, four values are written to main_jcp.aux for sm_jcp.tex: the theorem,
%      proposition and footnote counters (main.tex does not reset them at the SM), and
%      the REVTeX table-strut state (see sm_jcp.tex, ARSTRUT).
%
%  Build (no BibTeX; the bibliography is hand-written), from paper/:
%      pdflatex main_jcp ; pdflatex sm_jcp ; repeat both until neither asks for a rerun
%  or, with the checks against main.pdf:   python paper/pra_split.py --build
% =====================================================================================
\RequirePackage{xr-hyper}
\makeatletter
\AddToHook{begindocument/before}{%
  \externaldocument[][nocite]{sm_jcp}%
  \@for\PRA@l:=FirstPage,LastPage,LastBibItem\do{%
    \global\expandafter\let\csname r@\PRA@l\endcsname\relax}%
}
\def\PRA@smbegin{sm_begin.tex}
\def\PRA@export#1{%
  \immediate\write\@auxout{\string\newlabel{PRAsplit:#1}{{\the\value{#1}}{}{}{}{}}}}
\def\PRA@stop#1{%
  \PRA@export{theorem}\PRA@export{proposition}\PRA@export{footnote}%
  \immediate\write\@auxout{\string\newlabel{PRAsplit:arstrut}%
    {{\ifx\@arstrut\@arstrut@hline 1\else 0\fi}{}{}{}{}}}%
  \end{document}}
% --- 2026-11-08: the reference list of the main text --------------------------------------
% bibliography.tex lists all 142 works, those of the supplementary material included, and a
% hand-written thebibliography prints every entry it holds.  The main PDF now prints only the
% works the main text cites, in the order it first cites them -- the mechanism sm_jcp.tex has
% used for its own list since 2026-10-23, copied here: the order is read from the \citation
% records of this file's previous run (main_jcp.aux), each entry is taken, as tokens, from
% bibliography.tex at build time, and nothing is retyped.  On a first run (no order yet) the
% whole of bibliography.tex is printed, as before.
\newtoks\JCP@out
\def\JCP@keys{}
\def\JCP@citation#1{\@for\JCP@k:=#1\do{%
  \ifcsname JCP@seen@\JCP@k\endcsname\else
    \global\expandafter\let\csname JCP@seen@\JCP@k\endcsname\@empty
    \xdef\JCP@keys{\ifx\JCP@keys\@empty\else\JCP@keys,\fi\JCP@k}%
  \fi}}
\AddToHook{begindocument/before}{\let\citation\JCP@citation}
\long\def\JCP@grab#1\end{\JCP@split#1\bibitem\JCP@none\end}
\long\def\JCP@split#1\bibitem#2{%
  \ifx\JCP@key\@empty\else
    \expandafter\gdef\csname JCP@e@\JCP@key\endcsname{#1}\fi
  \def\JCP@key{#2}%
  \ifx\JCP@key\JCP@nonemark\expandafter\@gobble\else\expandafter\@firstofone\fi
  {\JCP@split}}
\def\JCP@nonemark{\JCP@none}
\def\JCP@item#1{\bibitem{#1}\csname JCP@e@#1\endcsname}
\def\JCP@bibfile{bibliography.tex}
\def\JCP@mainbib#1{%
  \ifx\JCP@keys\@empty
    \@latex@warning@no@line{main_jcp: no citation order yet -- the full list is printed; rerun}%
    \@iinput{#1}%
  \else
    \begingroup
      \def\JCP@key{}%
      \def\thebibliography##1{\JCP@grab}%
      \let\endthebibliography\relax
      \@iinput{#1}%
    \endgroup
    \global\JCP@out{}%
    \@for\JCP@k:=\JCP@keys\do{%
      \ifcsname JCP@e@\JCP@k\endcsname
        \global\JCP@out\expandafter{\the\expandafter\JCP@out
          \expandafter\JCP@item\expandafter{\JCP@k}}%
      \fi}%
    \begin{thebibliography}{99}\the\JCP@out\end{thebibliography}%
  \fi}
\def\PRA@iinput#1{\def\PRA@tmp{#1}%
  \ifx\PRA@tmp\PRA@smbegin\expandafter\PRA@stop
  \else\ifx\PRA@tmp\JCP@bibfile\expandafter\expandafter\expandafter\JCP@mainbib
  \else\expandafter\expandafter\expandafter\@iinput\fi\fi{#1}}
\AddToHook{begindocument/end}{%
  \def\input{\@ifnextchar\bgroup\PRA@iinput\@@input}}
\makeatother
\input{main.tex}
